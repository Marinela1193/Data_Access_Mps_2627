public static List<FlightPassengerDTO> searchFlightPassengers(String targetFlightCode) {
        List<FlightPassengerDTO> dtoList = new ArrayList<>();
        
        String sql = "SELECT f.flight_code, p.firstname, p.lastname, f.departure " +
                     "FROM flights f " +
                     "INNER JOIN tickets t ON f.flight_code = t.flight_code " +
                     "INNER JOIN passengers p ON t.passportno = p.passportno " +
                     "WHERE f.flight_code = ? AND t.date_of_cancellation IS NULL";

        try (Connection conn = ConnectionManager.getConnection();
             PreparedStatement pstm = conn.prepareStatement(sql)) {

            pstm.setString(1, targetFlightCode.trim().toUpperCase());

            try (ResultSet rs = pstm.executeQuery()) {
                while (rs.next()) {
                    String code = rs.getString("flight_code");
                    String fullName = rs.getString("firstname") + " " + rs.getString("lastname");
                    String depTime = rs.getString("departure");

                    // Aquí creas el objeto DTO con los datos de la fila actual
                    FlightPassengerDTO dto = new FlightPassengerDTO(code, fullName, depTime);
                    
                    // Y lo añadidas a tu lista
                    dtoList.add(dto);
                }
            }

        } catch (SQLException | ClassNotFoundException e) {
            System.out.println("Error en la consulta DTO: " + e.getMessage());
            LoggerService.log("Error en searchFlightPassengers: " + e.getMessage());
        }

        return dtoList;
    }

public static void deleteTicket(int ticketNumber, boolean force) {
    // 1. Si NO viene --force, pedimos confirmación por consola
    if (!force) {
        Scanner sc = new Scanner(System.in);
        System.out.print("Are you sure you want to delete ticket " + ticketNumber + "? (y/n): ");
        String answer = sc.nextLine().trim().toLowerCase();

        if (!answer.equals("y")) {
            System.out.println("Operation cancelled.");
            LoggerService.log("Delete cancelled by user for ticket " + ticketNumber);
            return; // Cortamos aquí y ni siquiera tocamos la BD
        }
    }

    // 2. Si venía --force o el usuario dijo "y", ejecutamos el DELETE
    String sql = "DELETE FROM tickets WHERE ticket_number = ?";

    try (Connection conn = ConnectionManager.getConnection();
         PreparedStatement pst = conn.prepareStatement(sql)) {

        pst.setInt(1, ticketNumber);
        int rows = pst.executeUpdate();

        if (rows > 0) {
            System.out.println("Ticket " + ticketNumber + " deleted successfully.");
            LoggerService.log("Ticket " + ticketNumber + " deleted.");
        } else {
            System.out.println("Error: Ticket " + ticketNumber + " not found.");
        }

    } catch (SQLException e) {
        System.out.println("Error deleting ticket: " + e.getMessage());
        LoggerService.log("Error deleting ticket: " + e.getMessage());
    }
}

public static void deletePassenger(String passportNo, boolean force) {
    if (!force) {
        Scanner sc = new Scanner(System.in);
        System.out.print("This will delete passenger " + passportNo + " and ALL their tickets. Continue? (y/n): ");
        String answer = sc.nextLine().trim().toLowerCase();
        if (!answer.equals("y")) {
            System.out.println("Operation cancelled.");
            return;
        }
    }

    String sqlDeleteTickets = "DELETE FROM tickets WHERE passportno = ?";
    String sqlDeletePassenger = "DELETE FROM passengers WHERE passportno = ?";

    try (Connection conn = ConnectionManager.getConnection();
         PreparedStatement pstTickets = conn.prepareStatement(sqlDeleteTickets);
         PreparedStatement pstPassenger = conn.prepareStatement(sqlDeletePassenger)) {

        conn.setAutoCommit(false); // 1. Iniciamos transacción

        try {
            // 2. PRIMERO borramos los HIJOS (tickets)
            pstTickets.setString(1, passportNo);
            int ticketsDeleted = pstTickets.executeUpdate();

            // 3. DESPUÉS borramos al PADRE (passenger)
            pstPassenger.setString(1, passportNo);
            int passengerDeleted = pstPassenger.executeUpdate();

            if (passengerDeleted > 0) {
                conn.commit(); // 4. Todo ok -> guardamos ambos borrados
                System.out.println("Passenger deleted (along with " + ticketsDeleted + " tickets).");
                LoggerService.log("Deleted passenger " + passportNo + " and " + ticketsDeleted + " tickets.");
            } else {
                conn.rollback(); // No existía ese pasajero
                System.out.println("Error: Passenger " + passportNo + " not found.");
            }

        } catch (SQLException e) {
            conn.rollback(); // Si falla algo, deshacemos todo
            System.out.println("Error deleting passenger. Transaction rolled back: " + e.getMessage());
            LoggerService.log("Rollback deleting passenger: " + e.getMessage());
        } finally {
            conn.setAutoCommit(true);
        }

    } catch (SQLException e) {
        e.printStackTrace();
    }
}

public static void exportFlightsToFile(String filename) {
    String sql = "SELECT flight_code, duration, distance FROM flights ORDER BY flight_code";

    try (Connection conn = ConnectionManager.getConnection();
         Statement stmt = conn.createStatement();
         ResultSet rs = stmt.executeQuery(sql);
         PrintWriter pw = new PrintWriter(new BufferedWriter(new FileWriter(filename, false)))) {

        // 1. Escribimos una cabecera (opcional, pero queda muy bien)
        pw.println("FLIGHT_CODE;DURATION;DISTANCE");

        // 2. Recorremos la tabla fila por fila y escribimos en el fichero
        int count = 0;
        while (rs.next()) {
            String code = rs.getString("flight_code");
            int duration = rs.getInt("duration");
            int distance = rs.getInt("distance");

            pw.println(code + ";" + duration + ";" + distance);
            count++;
        }

        System.out.println("Exported " + count + " flights to " + filename);
        LoggerService.log("Exported " + count + " flights to " + filename);

    } catch (SQLException | IOException e) {
        System.out.println("Error exporting flights: " + e.getMessage());
        LoggerService.log("Error exporting flights: " + e.getMessage());
    }
}
